interface TaskTreeNode extends Task {
  children: TaskTreeNode[];
  depth: number;
  isLeaf: boolean;
}

function buildTaskTree(tasks: Task[]): TaskTreeNode[] {
  const map = new Map<string, TaskTreeNode>();
  const roots: TaskTreeNode[] = [];

  tasks.forEach(t => map.set(t.id, { ...t, children: [], depth: 0, isLeaf: true }));
  tasks.forEach(t => {
    if (t.parentId && map.has(t.parentId)) {
      map.get(t.parentId)!.children.push(map.get(t.id)!);
      map.get(t.parentId)!.isLeaf = false;
    } else {
      roots.push(map.get(t.id)!);
    }
  });

  // 閫掑綊璁＄畻 depth
  function assignDepth(nodes: TaskTreeNode[], d: number) {
    nodes.forEach(n => { n.depth = d; assignDepth(n.children, d + 1); });
  }
  assignDepth(roots, 0);

  return roots;
}
